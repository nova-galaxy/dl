import os
import json
import random
import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf

from tensorflow.keras.applications.vgg16 import VGG16, preprocess_input
from tensorflow.keras.preprocessing import image
from sklearn.linear_model import LogisticRegression
from sklearn.metrics import (
    accuracy_score, precision_score, recall_score,
    f1_score, confusion_matrix, ConfusionMatrixDisplay
)

# 1. Dataset paths
DATASET_DIR = "/content/ImageNet"
JSON_PATH = "/content/ImageNet/imagenet_class_index.json"

# 2. Load ImageNet class names
with open(JSON_PATH, "r") as f:
    data = json.load(f)

class_names = {}

for key, value in data.items():
    if isinstance(value, list):
        class_names[value[0]] = value[1]
        class_names[key] = value[1]
    else:
        class_names[key] = value

# 3. Read images from existing class folders
class_images = {}

for root, dirs, files in os.walk(DATASET_DIR):
    folder = os.path.relpath(root, DATASET_DIR).split(os.sep)[0]

    if folder not in class_names:
        continue

    for file in files:
        if file.lower().endswith((".jpg", ".jpeg", ".png")):
            path = os.path.join(root, file)
            class_images.setdefault(folder, []).append(path)

# 4. Select up to 8 classes with at least 2 images
usable = [
    c for c in class_images
    if len(class_images[c]) >= 2
]

random.seed(42)
random.shuffle(usable)
usable = usable[:8]

if len(usable) < 2:
    raise ValueError("Need at least 2 classes with 2 images each.")

# 5. Create training and testing sets
train_paths, test_paths = [], []
train_labels, test_labels = [], []

for c in usable:
    paths = class_images[c].copy()
    random.shuffle(paths)

    # Limit images to keep the program manageable
    paths = paths[:20]

    if len(paths) < 2:
        continue

    split = max(1, int(0.8 * len(paths)))
    split = min(split, len(paths) - 1)

    train_paths += paths[:split]
    test_paths += paths[split:]

    train_labels += [class_names[c]] * split
    test_labels += [class_names[c]] * (len(paths) - split)

print("Classes:", [class_names[c] for c in usable])
print("Training images:", len(train_paths))
print("Testing images:", len(test_paths))

# 6. Load pretrained VGG16 without its classifier
base_model = VGG16(
    weights="imagenet",
    include_top=False,
    pooling="avg"
)

# Global Average Pooling produces 512 features per image
print("Feature extractor loaded")

# 7. Extract features
def extract_features(paths):
    imgs = []

    for path in paths:
        img = image.load_img(path, target_size=(224, 224))
        img = image.img_to_array(img)
        imgs.append(img)

    imgs = np.array(imgs)
    imgs = preprocess_input(imgs)

    features = base_model.predict(imgs, verbose=0)
    return features

X_train = extract_features(train_paths)
X_test = extract_features(test_paths)

print("Training feature shape:", X_train.shape)
print("Testing feature shape:", X_test.shape)

# 8. Train a new classifier
classifier = LogisticRegression(max_iter=1000)
classifier.fit(X_train, train_labels)

# 9. Predict unseen test images
predictions = classifier.predict(X_test)

# 10. Calculate performance metrics
print("Accuracy:", accuracy_score(test_labels, predictions))
print("Precision:", precision_score(
    test_labels, predictions, average="macro", zero_division=0
))
print("Recall:", recall_score(
    test_labels, predictions, average="macro", zero_division=0
))
print("F1-score:", f1_score(
    test_labels, predictions, average="macro", zero_division=0
))

# 11. Display confusion matrix
classes = classifier.classes_

cm = confusion_matrix(
    test_labels, predictions, labels=classes
)

display = ConfusionMatrixDisplay(
    confusion_matrix=cm,
    display_labels=classes
)

display.plot(xticks_rotation=90)
plt.title("Confusion Matrix")
plt.tight_layout()
plt.show()
